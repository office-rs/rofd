```rust
fn parse_vertices(params: &[(String, String)]) -> Vec<Point> {
    params.iter().find(|(k, _)| k == "Vertices")
        .and_then(|(_, v)| {
            let n: Vec<f64> = v.split_whitespace().filter_map(|t| t.parse().ok()).collect();
            (n.len() % 2 == 0).then(|| n.chunks(2).map(|c| Point { x: c[0], y: c[1] }).collect())
        })
        .unwrap_or_default()
}
```

```rust
pub fn ellipse_path(r: &Rect) -> PathData {
    let (cx, cy) = (r.w / 2.0, r.h / 2.0);
    let (rx, ry) = (r.w / 2.0, r.h / 2.0);
    PathData { commands: vec![
        PathCommand::M(cx + rx, cy),
        PathCommand::A(rx, ry, 0.0, 0.0, 1.0, cx, cy + ry),
        PathCommand::A(rx, ry, 0.0, 0.0, 1.0, cx - rx, cy),
        PathCommand::A(rx, ry, 0.0, 0.0, 1.0, cx, cy - ry),
        PathCommand::A(rx, ry, 0.0, 0.0, 1.0, cx + rx, cy),
        PathCommand::Z,
    ]}
}
```

```rust
pub fn arrow_path(r: &Rect) -> PathData {
    let (w, h) = (r.w, r.h);
    let head = w.min(h).max(1.0) * 0.25;
    // 主线 (0,0)->(w,h)；三角头在 (w,h)，沿主线方向
    let angle = (h).atan2(w);
    let (dx, dy) = (angle.cos() * head, angle.sin() * head);
    let (nx, ny) = (-angle.sin() * head, angle.cos() * head);
    PathData { commands: vec![
        PathCommand::M(0.0, 0.0), PathCommand::L(w, h),
        PathCommand::M(w - dx + nx, h - dy + ny), PathCommand::L(w, h), PathCommand::L(w - dx - nx, h - dy - ny),
        PathCommand::Z,
    ]}
}
```

```rust
pub fn polygon_path(points: &[Point]) -> PathData {
    let mut cmds = Vec::with_capacity(points.len() + 1);
    if let Some(p0) = points.first() {
        cmds.push(PathCommand::M(p0.x, p0.y));
        for p in &points[1..] { cmds.push(PathCommand::L(p.x, p.y)); }
        cmds.push(PathCommand::Z);
    }
    PathData { commands: cmds }
}

pub fn polyline_path(points: &[Point]) -> PathData {
    let mut cmds = Vec::with_capacity(points.len());
    if let Some(p0) = points.first() {
        cmds.push(PathCommand::M(p0.x, p0.y));
        for p in &points[1..] { cmds.push(PathCommand::L(p.x, p.y)); }
    }
    PathData { commands: cmds }
}

/// Squiggly 波浪 Q 曲线：在 quad_points 对角矩形内交替上下波浪。
pub fn squiggly_path(p0: Point, p1: Point) -> PathData {
    let mut cmds = vec![PathCommand::M(p0.x, p0.y)];
    let steps = 20;
    let dx = (p1.x - p0.x) / steps as f64;
    let amp = (p1.y - p0.y).abs() / 2.0;
    for i in 0..steps {
        let x0 = p0.x + dx * i as f64;
        let x1 = p0.x + dx * (i as f64 + 1.0);
        let y_mid = if i % 2 == 0 { p0.y - amp } else { p0.y + amp };
        cmds.push(PathCommand::Q(x0, y_mid, x1, p0.y));
    }
    PathData { commands: cmds }
}
```

```rust
fn draw_markup(painter: &mut Painter, kind: AnnotationKind, quad_points: &[Point], color: &Color, base: Affine) {
    let peniko_color = color::to_peniko(color).with_alpha(if matches!(kind, AnnotationKind::Highlight) { 0.38 } else { 1.0 });
    for (p0, p1) in quad_point_pairs(quad_points) {
        match kind {
            AnnotationKind::Squiggly => {
                // 波浪 Q 曲线
                let mut path = BezPath::new();
                path.move_to((p0.x, p0.y));
                let steps = 20;
                let dx = (p1.x - p0.x) / steps as f64;
                let amp = 1.0;
                for i in 0..steps {
                    let xm = p0.x + dx * (i as f64 + 0.5);
                    let x1 = p0.x + dx * (i as f64 + 1.0);
                    let ym = if i % 2 == 0 { p0.y - amp } else { p0.y + amp };
                    path.quad_to((xm, ym), (x1, p0.y));
                }
                painter.stroke(&path, peniko_color, 1.0, base);
            }
            _ => {
                // 现有半透明矩形（Highlight/Underline/Strikeout）
                let r = KurboRect::new(p0.x, p0.y, p1.x, p1.y);
                painter.fill(&r, peniko_color, base);
            }
        }
    }
}
```
